Option Explicit

Function split_csv(fn, chunkSize)
    On Error Resume Next
    Dim data() As Variant
    Dim i, j, line, header, outfile
    Dim rowcnt As Long
    Dim chunkcnt As Long
    chunkcnt = 0
    rowcnt = 0
    Debug.Print "reading file : ", fn
    Open fn For Input As #1
    Line Input #1, line
    If line = "" Then
        Exit Function
    End If
    header = Replace(line, "ï»¿", "")
    Do While Not EOF(1)
        DoEvents
        ReDim data(1 To chunkSize, 1 To 1)
        chunkcnt = chunkcnt + 1
        outfile = Replace(fn, ".csv", "-" + Right("000" & chunkcnt, 3)) + ".csv"
        Open outfile For Output As #2
        Print #2, header
        j = 0
        For i = 1 To chunkSize
            If EOF(1) Then
                Exit For
            End If
            Line Input #1, line
            If line = "" Then Exit For
            Print #2, line
            j = j + 1
            DoEvents
        Next i
        Close #2
        rowcnt = rowcnt + j
    Loop
    Close #1
    split_csv = rowcnt
End Function

Function join_csvfiles(outfile, filespec)
    Dim i, j, line, header, spec
    Dim cnt As Long
    On Error Resume Next
    spec = Dir(filespec)
    cnt = 0
    header = ""
    Open outfile For Output As #2
    While spec <> ""
        Debug.Print "reading file : ", spec
        Open spec For Input As #1
        Line Input #1, line
        If cnt = 0 Then
            header = Replace(line, "ï»¿", "")
            Debug.Print header
            Print #2, header
        End If
        Do While Not EOF(1)
            DoEvents
            If EOF(1) Then
                Exit Do
            End If
            Line Input #1, line
            If line = "" Then Exit Do
            Print #2, line
        Loop
        Close #1
        spec = Dir()
        cnt = cnt + 1
    Wend
    Close #2
    join_csvfiles = cnt
End Function

Sub split_largefile()
    On Error Resume Next
    Dim fn
    Dim chunkSize As Long       ' Adjust chunk size as needed
    Dim rowcnt As Long
    ChDir "c:\temp"
    fn = "zz.csv"
    chunkSize = 100000
    rowcnt = split_csv(fn, chunkSize) ' returns 1845970
    MsgBox "large file successsfully splitted, rows count = " & rowcnt, vbInformation, "Split Large CSV file"
End Sub

Sub merge_largefile()
    On Error Resume Next
    Dim fn, filespec
    Dim rowcnt As Long
    ChDir "c:\temp"
    fn = "zz.csv"
    filespec = "zz-*.csv"
    rowcnt = join_csvfiles(fn, filespec) ' returns 1845970
    MsgBox "csv file successsfully merged as " + fn + " , rows count = " & rowcnt, vbInformation, "Merge CSV files"
End Sub
